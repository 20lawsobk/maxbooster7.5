import { useEffect, useMemo, useState } from "react";
import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import { getCsrfTokenFromCookie } from "@/lib/queryClient";
import { useToast } from "@/hooks/use-toast";
import { Alert, AlertDescription } from "@/components/ui/alert";
import { Button } from "@/components/ui/button";
import {
  Card,
  CardContent,
  CardDescription,
  CardHeader,
  CardTitle,
} from "@/components/ui/card";
import { Label } from "@/components/ui/label";
import { Textarea } from "@/components/ui/textarea";
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from "@/components/ui/select";
import { AlertCircle, Loader2, RotateCcw, Save, Sparkles } from "lucide-react";
 
export interface GenerationQualitySettings {
  creativeDirection: string;
  avoidTopics: string[];
  textVariants: number;
  imageMaxEdge: 1080 | 2048;
  audioFormat: "wav";
  audioSampleRate: 44100;
  audioBitDepth: 16 | 24;
  videoOutputResolution: "720p" | "1080p";
  videoFps: 24 | 30;
  videoSampleCount: number;
}
 
const DEFAULTS: GenerationQualitySettings = {
  creativeDirection: "",
  avoidTopics: [],
  textVariants: 3,
  imageMaxEdge: 2048,
  audioFormat: "wav",
  audioSampleRate: 44100,
  audioBitDepth: 24,
  videoOutputResolution: "1080p",
  videoFps: 30,
  videoSampleCount: 1,
};
 
const QUERY_KEY = ["/api/admin/beat-money-loop/generation-quality"];
 
async function readSettings(): Promise<GenerationQualitySettings> {
  const response = await fetch(QUERY_KEY[0], { credentials: "include" });
  if (!response.ok) throw new Error(`Could not load settings (${response.status})`);
  const result = await response.json();
  if (!result?.settings || typeof result.settings !== "object") {
    throw new Error("The server returned an invalid generation settings response");
  }
  return { ...DEFAULTS, ...result.settings };
}
 
export function GenerationQualityPanel() {
  const { toast } = useToast();
  const queryClient = useQueryClient();
  const settingsQuery = useQuery({
    queryKey: QUERY_KEY,
    queryFn: readSettings,
    staleTime: 5_000,
    retry: 1,
  });
  const [settings, setSettings] = useState(DEFAULTS);
  const [avoidText, setAvoidText] = useState("");
 
  useEffect(() => {
    if (!settingsQuery.data) return;
    setSettings(settingsQuery.data);
    setAvoidText(settingsQuery.data.avoidTopics.join("\n"));
  }, [settingsQuery.data]);
 
  const saved = settingsQuery.data ?? DEFAULTS;
  const normalized = useMemo(
    () => ({
      ...settings,
      avoidTopics: avoidText
        .split("\n")
        .map((item) => item.trim())
        .filter(Boolean),
    }),
    [settings, avoidText],
  );
  const dirty = JSON.stringify(normalized) !== JSON.stringify(saved);
 
  const saveMutation = useMutation({
    mutationFn: async () => {
      const csrf = getCsrfTokenFromCookie();
      const response = await fetch(QUERY_KEY[0], {
        method: "PUT",
        credentials: "include",
        headers: {
          "Content-Type": "application/json",
          ...(csrf ? { "x-csrf-token": csrf } : {}),
        },
        body: JSON.stringify(normalized),
      });
      const body = await response.json().catch(() => ({}));
      if (!response.ok) {
        const detail =
          body.issues?.map((issue: { message?: string }) => issue.message).filter(Boolean).join("; ") ||
          body.error ||
          `Could not save settings (${response.status})`;
        throw new Error(detail);
      }
      return body.settings as GenerationQualitySettings;
    },
    onSuccess: async (savedSettings) => {
      queryClient.setQueryData(QUERY_KEY, savedSettings);
      setSettings(savedSettings);
      setAvoidText(savedSettings.avoidTopics.join("\n"));
      toast({ title: "Generation settings saved", description: "Applied to platform generation requests." });
    },
    onError: (error: Error) =>
      toast({ title: "Could not save generation settings", description: error.message, variant: "destructive" }),
  });
 
  const reset = () => {
    setSettings(saved);
    setAvoidText(saved.avoidTopics.join("\n"));
  };
 
  if (settingsQuery.isLoading) {
    return (
      <Card>
        <CardContent className="flex items-center gap-2 py-8 text-sm text-muted-foreground">
          <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" />
          Loading generation settings…
        </CardContent>
      </Card>
    );
  }
 
  if (settingsQuery.isError) {
    return (
      <Alert variant="destructive">
        <AlertCircle className="h-4 w-4" />
        <AlertDescription className="flex items-center justify-between gap-4">
          <span>{(settingsQuery.error as Error).message}</span>
          <Button variant="outline" size="sm" onClick={() => settingsQuery.refetch()}>
            Retry
          </Button>
        </AlertDescription>
      </Alert>
    );
  }
 
  return (
    <Card data-testid="generation-quality-panel">
      <CardHeader>
        <CardTitle className="flex items-center gap-2">
          <Sparkles className="h-5 w-5" aria-hidden="true" />
          Platform generation quality
        </CardTitle>
        <CardDescription>
          Shared defaults for text, image, audio, and video generation. A specific
          request’s explicit settings take priority over these defaults.
        </CardDescription>
      </CardHeader>
      <CardContent className="space-y-6">
        <div className="grid gap-5 md:grid-cols-2">
          <div className="space-y-2">
            <Label htmlFor="generation-direction">Creative direction</Label>
            <Textarea
              id="generation-direction"
              value={settings.creativeDirection}
              maxLength={600}
              onChange={(event) =>
                setSettings((current) => ({
                  ...current,
                  creativeDirection: event.target.value,
                }))
              }
              placeholder="Optional guidance applied to every generation type"
              rows={4}
              aria-describedby="generation-direction-help"
            />
            <p id="generation-direction-help" className="text-xs text-muted-foreground">
              Up to 600 characters. This is creative context, not a guarantee of engagement or sales.
            </p>
          </div>
          <div className="space-y-2">
            <Label htmlFor="generation-avoid">Avoid topics or treatments</Label>
            <Textarea
              id="generation-avoid"
              value={avoidText}
              onChange={(event) => setAvoidText(event.target.value)}
              placeholder={"One item per line\nMaximum 10 items"}
              rows={4}
              aria-describedby="generation-avoid-help"
            />
            <p id="generation-avoid-help" className="text-xs text-muted-foreground">
              One per line; each item is limited to 80 characters.
            </p>
          </div>
        </div>
 
        <div className="grid gap-5 sm:grid-cols-2 lg:grid-cols-3">
          <SettingSelect
            label="Text alternatives"
            value={String(settings.textVariants)}
            onValueChange={(value) => setSettings((v) => ({ ...v, textVariants: Number(value) }))}
            options={[1, 2, 3, 4, 5].map((value) => [String(value), `${value} candidates`])}
            help="Ranks actual generated copy; does not predict performance."
          />
          <SettingSelect
            label="Image maximum edge"
            value={String(settings.imageMaxEdge)}
            onValueChange={(value) => setSettings((v) => ({ ...v, imageMaxEdge: Number(value) as 1080 | 2048 }))}
            options={[["1080", "1080 px"], ["2048", "2048 px"]]}
            help="Preserves each platform’s aspect ratio."
          />
          <SettingSelect
            label="Audio export"
            value={String(settings.audioBitDepth)}
            onValueChange={(value) => {
              setSettings((v) => ({
                ...v,
                audioFormat: "wav",
                audioSampleRate: 44100,
                audioBitDepth: Number(value) as 16 | 24,
              }));
            }}
            options={[
              ["24", "WAV · 44.1 kHz · 24-bit"],
              ["16", "WAV · 44.1 kHz · 16-bit"],
            ]}
            help="Lossless WAV export; dedicated music rendering validates sample rate and bit depth."
          />
          <SettingSelect
            label="Video resolution"
            value={settings.videoOutputResolution}
            onValueChange={(value) => setSettings((v) => ({ ...v, videoOutputResolution: value as "720p" | "1080p" }))}
            options={[["720p", "720p"], ["1080p", "1080p"]]}
            help="Applied as the long edge while retaining platform orientation."
          />
          <SettingSelect
            label="Video frame rate"
            value={String(settings.videoFps)}
            onValueChange={(value) => setSettings((v) => ({ ...v, videoFps: Number(value) as 24 | 30 }))}
            options={[["24", "24 fps"], ["30", "30 fps"]]}
            help="The native renderer validates frame and pixel budgets."
          />
          <SettingSelect
            label="Video candidates"
            value={String(settings.videoSampleCount)}
            onValueChange={(value) => setSettings((v) => ({ ...v, videoSampleCount: Number(value) }))}
            options={[1, 2, 3, 4].map((value) => [String(value), `${value} render${value === 1 ? "" : "s"}`])}
            help="More renders increase compute time and storage use."
          />
        </div>
 
        {avoidText.split("\n").map((item) => item.trim()).filter(Boolean).length > 10 && (
          <Alert variant="destructive">
            <AlertCircle className="h-4 w-4" />
            <AlertDescription>Enter no more than 10 avoid items before saving.</AlertDescription>
          </Alert>
        )}
 
        <div className="flex flex-wrap items-center justify-end gap-2 border-t pt-4">
          <Button variant="outline" onClick={reset} disabled={!dirty || saveMutation.isPending}>
            <RotateCcw className="mr-2 h-4 w-4" aria-hidden="true" />
            Reset changes
          </Button>
          <Button
            onClick={() => saveMutation.mutate()}
            disabled={!dirty || saveMutation.isPending || normalized.avoidTopics.length > 10 || normalized.avoidTopics.some((item) => item.length > 80)}
          >
            {saveMutation.isPending ? (
              <Loader2 className="mr-2 h-4 w-4 animate-spin" aria-hidden="true" />
            ) : (
              <Save className="mr-2 h-4 w-4" aria-hidden="true" />
            )}
            Save settings
          </Button>
        </div>
      </CardContent>
    </Card>
  );
}
 
function SettingSelect({
  label,
  value,
  onValueChange,
  options,
  help,
}: {
  label: string;
  value: string;
  onValueChange: (value: string) => void;
  options: [string, string][];
  help: string;
}) {
  const id = `generation-setting-${label.toLowerCase().replace(/[^a-z0-9]+/g, "-")}`;
  return (
    <div className="space-y-2">
      <Label htmlFor={id}>{label}</Label>
      <Select value={value} onValueChange={onValueChange}>
        <SelectTrigger id={id}>
          <SelectValue />
        </SelectTrigger>
        <SelectContent>
          {options.map(([optionValue, optionLabel]) => (
            <SelectItem key={optionValue} value={optionValue}>
              {optionLabel}
            </SelectItem>
          ))}
        </SelectContent>
      </Select>
      <p className="text-xs text-muted-foreground">{help}</p>
    </div>
  );
}
